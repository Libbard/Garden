class Member:
    def __init__(self, username, fullname):
        self.username = username
        self.fullname = fullname
        self.is_active = True

    # Method with parameters
    def activate(self, yesno):
        """ True for active, False to make inactive """
        self.is_active = yesno

new_guy = Member('Rambo', 'Rocco Moe')
new_guy.activate(False) # self is passed automatically
# Alternative way to call:
# Member.activate(new_guy, False)

import datetime as dt

class Member:
    free_days = 0

    @classmethod
    def setfreedays(cls, days):
        cls.free_days = days

    @staticmethod
    def currenttime():
        now = dt.datetime.now()
        return f"{now:%I:%M %p}"

# Calling class method
Member.setfreedays(30)

# Calling static method
print(Member.currenttime())

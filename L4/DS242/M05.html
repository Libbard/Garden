keys = ['a', 'b', 'c']
# Without value (defaults to None)
dict1 = dict.fromkeys(keys)
print(dict1) # {'a': None, 'b': None, 'c': None}

# With a specific value
dict2 = dict.fromkeys(keys, 0)
print(dict2) # {'a': 0, 'b': 0, 'c': 0}

products = {
    'RB00111': {'name': 'Ray-Ban', 'price': 112.98},
    'DWC0317': {'name': 'Drone', 'price': 72.95}
}

# Looping through nested dictionary
for prod_id, details in products.items():
    print(f"ID: {prod_id} | Name: {details['name']} | Price: ${details['price']}")

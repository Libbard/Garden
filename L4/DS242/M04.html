أهداف التعلم

Lists (القوائم)Dictionaries (القواميس)
python 
people = {
    'htanaka': 'Haru Tanaka',
    'ppatel': 'Priya Patel',
    'zmin': 'Zhang Min'
}
python 
person = 'zmin'
print(people[person]) # Prints 'Zhang Min'

print(len(people)) # Prints the total number of items

print('hajackson' in people) # Prints True
python 
person = 'schmeedledorp'
print(people.get(person)) # Prints: None
python 
# Change existing key
people.update({'hrjackson' : 'Henrietta Jackson'})

# Add new key
people.update({'wwiggins' : 'Wanda Wiggins'})
python 
# Looping through keys to get values
for person in people:
    print(people[person])

# Looping directly through values
for person in people.values():
    print(person)

import datetime as dt
today = dt.date.today()
birthdate = dt.date(2000, 1, 31)

# Duration between dates as a timedelta
delta_age = today - birthdate

# Extract days as a number
days_old = delta_age.days

# Floor divide by 365 to get years
years = days_old // 365

# Remainder divided by 30 for approximate months
months = (days_old % 365) // 30

print(f"You are {years} years and {months} months old.")

def csv_reader_dataset(filepaths):
    dataset = tf.data.Dataset.list_files(filepaths).repeat()
    dataset = dataset.interleave(lambda filepath: tf.data.TextLineDataset(filepath).skip(1), num_parallel_calls=tf.data.AUTOTUNE)
    dataset = dataset.shuffle(buffer_size=10000)
    return dataset
